# Third-party dependencies of the agent tree. Exact pins: an unpinned >= produced a
# ResolutionTooDeep failure before any code ran (agent-design.md §21).
#
# Install once per session, before %run-ing anything under notebooks/agents/:
#
#     %pip install -r ./requirements.txt
#     dbutils.library.restartPython()
#
# restartPython wipes the namespace, which is why no notebook installs anything itself - a %pip
# inside llm/gateway would reset every caller that %run it (agent-design.md §3.3). On serverless,
# pointing the notebook Environment panel at this file does the same thing without the restart.
#
# The job that eventually runs these notebooks declares the same pins in its environment spec; the
# two lists have to stay equal, and this file is the one to copy from.
#
# pydantic is deliberately not pinned here: langchain-core already requires v2, which is what the
# schemas in llm/parsing need (model_json_schema, model_validate_json), and a second pin on it is
# how a resolution conflict starts. tiktoken is not needed either - the embedding client is built
# with check_embedding_ctx_length=False precisely so nothing tries to tokenize locally.

langchain-openai==1.6.2
langchain-core==1.6.3
openai==3.13.0
